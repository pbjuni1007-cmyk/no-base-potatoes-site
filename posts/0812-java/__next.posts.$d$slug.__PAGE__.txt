// 2. Service 메서드: 최소 규칙을 확인하고 PENDING 예약 생성
public ReservationResponse create(ReservationRequest request) {
    if (request.partySize() <= 0) {
        throw new IllegalArgumentException("예약 인원은 1명 이상이어야 합니다.");
    }

    Reservation pending = new Reservation(
            request.id(), request.partySize(), "PENDING");
    Reservation saved = reservationRepository.save(pending);

    return new ReservationResponse(
            saved.id(), saved.partySize(), saved.status());
}
